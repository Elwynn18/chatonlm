1// Turns a title into a URL slug.
2export function slugify(title, max = 60) {
3  const slug = title
4    .normalize("NFKD")
5    .toLowerCase()
6    .replace(/[^a-z0-9]+/g, "-")
7    .replace(/^-+|-+$/g, "");
8  if (slug.length <= max)
9    return slug;
10  const cut = slug.slice(0, max);
11  const dash = cut.lastIndexOf("-");
12  return dash > 0 ? cut.slice(0, dash) : cut;
13}
